\begin{lemma}[Various Truths]\label{lem:truths}
    \begin{enumerate}
        \item If $K=\{\underline{v}\}$ then $K$ is dependent iff $\underline{v}=0$.
        \item If $K$ is independent and $T \subseteq K$, then $T$ is independent.
        \item If $K$ is dependent and $T \supseteq K$, then $T$ is dependent.
        \item If $K$ is independent and $\underline{v} \notin K$ is not dependent on $K$, then $K \cup \{\underline{v}\}$ is independent.
        \item If $K$ is independent and $\underline{v} \notin K$ is dependent on $K$, then $K \cup \{\underline{v}\}$ is dependent.
        \item If $\mathrm{Sp}(K \cup \{\underline{v}\}) = \mathrm{Sp}(K)$, then $\underline{v}$ is dependent on $K$.
        \item If $\underline{v}$ is dependent on $K$, then $\mathrm{Sp}(K \cup \{\underline{v}\}) = \mathrm{Sp}(K)$.
    \end{enumerate}

\end{lemma}
